            count = len(result["findings"])
            st.error(

                with st.expander(f"{finding['method']}{page_label}"):
                    st.code(finding["text"], language=None)

        else:  # status == "error"
            st.warning(
                f"⚠ **{result['filename']}** — Could not scan: {result.get('error', 'Unknown error')}"
            )
